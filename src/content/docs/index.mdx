import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

```jsx
import { Reveal } from "@awesome-reveal/react";
import { slide } from "@awesome-reveal/react/keyframes";

<Reveal keyframes={slide({ direction: "left" })}>
  <p>I gently appear as I enter the viewport</p>
</Reveal>;
```

## Features

<CardGrid>
  <Card title="No dependencies" icon="puzzle">
    The packages have no third-party dependencies. Each keyframes factory is a
    small function, so your bundle includes only the factories that you import.
  </Card>
  <Card title="Native APIs" icon="rocket">
    Intersection Observer detects the reveal. Web Animations runs it on the
    compositor where possible.
  </Card>
  <Card title="Accessible" icon="approve-check">
    It respects `prefers-reduced-motion` by default. Hidden content stays in the
    accessibility tree.
  </Card>
  <Card title="SSR and Server Components" icon="laptop">
    Content renders hidden on the server. You can create keyframes in React
    Server Components.
  </Card>
  <Card title="TypeScript" icon="seti:typescript">
    Strict types for the options, the elements and the keyframes.
  </Card>
  <Card title="Many frameworks" icon="setting">
    `@awesome-reveal/core` has no framework code. Each framework gets a small
    adapter on top of it.
  </Card>
</CardGrid>
