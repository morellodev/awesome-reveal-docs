import { Cascade } from "@awesome-reveal/astro";

<Cascade as="ul" class="features not-content" interval={100}>
  <li>
    <h2>Native browser APIs</h2>
    <p>
      Intersection Observer detects when an element enters the viewport. The Web
      Animations API animates it. The browser loads no animation library.
    </p>
  </li>
  <li>
    <h2>No flash of content</h2>
    <p>
      The server renders the content hidden. The content does not show before
      the animation starts. If the script fails, the content shows after 5
      seconds.
    </p>
  </li>
  <li>
    <h2>Reduced motion support</h2>
    <p>
      When the operating system requests reduced motion, the content shows at
      once, with no animation.
    </p>
  </li>
  <li>
    <h2>One core, four packages</h2>
    <p>
      The React, Vue, Astro and JavaScript packages share one core and the same
      options. Each package has TypeScript types.
    </p>
  </li>
</Cascade>
